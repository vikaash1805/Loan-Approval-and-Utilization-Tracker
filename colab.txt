class IdentityCardData(BaseModel):
    card_type: str = Field(description="Type of identity card: Aadhaar or PAN")
    name: Optional[str] = Field(None, description="Full name exactly as printed on the card")
    aadhaar_number: Optional[str] = Field(None, description="12-digit Aadhaar number (null if PAN)")
    pan_number: Optional[str] = Field(None, description="10-character PAN number (null if Aadhaar)")
    dob: Optional[str] = Field(None, description="Date of birth in DD/MM/YYYY or YYYY format")
    gender: Optional[str] = Field(None, description="Gender (Male, Female, Transgender)")
    is_valid_id: bool = Field(True, description="True if valid genuine Indian identity document")

def normalize_name(value: Optional[str]) -> str:
    """Removes special characters, extra spaces, and converts name to uppercase."""
    if not value:
        return ""
    val = str(value).upper().strip()
    val = re.sub(r"[^A-Z\s]", " ", val)
    return re.sub(r"\s+", " ", val).strip()

def parse_date_components(date_str: Optional[str]) -> Optional[Tuple[str, Optional[str], Optional[str]]]:
    """Extracts (YYYY, MM, DD) components from varied date formats."""
    if not date_str:
        return None
    cleaned = str(date_str).strip().replace(".", "/").replace("-", "/")
    parts = [p.strip() for p in cleaned.split("/") if p.strip()]

    if len(parts) == 3:
        if len(parts[0]) == 4:
            return (parts[0], parts[1].zfill(2), parts[2].zfill(2))
        elif len(parts[2]) == 4:
            return (parts[2], parts[1].zfill(2), parts[0].zfill(2))
        elif len(parts[2]) == 2:
            base_year = "19" if parts[2].isdigit() and int(parts[2]) > 30 else "20"
            yyyy = base_year + parts[2]
            return (yyyy, parts[1].zfill(2), parts[0].zfill(2))
    elif len(parts) == 1 and len(parts[0]) == 4:
        return (parts[0], None, None)
    return None

def normalize_dob(value: Optional[str]) -> str:
    """Standardizes date of birth into DD/MM/YYYY format or YYYY."""
    comps = parse_date_components(value)
    if comps:
        yyyy, mm, dd = comps
        if mm and dd:
            return f"{dd}/{mm}/{yyyy}"
        return yyyy
    return str(value or "").strip()

def normalize_aadhaar(value: Optional[str]) -> str:
    """Extracts only 12 digits, removing spaces and delimiters."""
    if not value:
        return ""
    return re.sub(r"\D", "", str(value))

def normalize_pan(value: Optional[str]) -> str:
    """Normalizes PAN to uppercase 10-character alphanumeric."""
    if not value:
        return ""
    return re.sub(r"[^A-Za-z0-9]", "", str(value)).upper()

def normalize_gender(value: Optional[str]) -> str:
    """Standardizes gender string into MALE, FEMALE, or TRANSGENDER."""
    if not value:
        return ""
    val = str(value).upper().strip()
    if "FEM" in val or val == "F":
        return "FEMALE"
    if "MALE" in val or val == "M":
        return "MALE"
    if "TRANS" in val or "OTHER" in val or val == "T":
        return "TRANSGENDER"
    return val

def sanitize_extracted_payload(raw_data: Dict[str, Any]) -> Dict[str, Any]:
    """Complete sanitization pipeline for all extracted text fields."""
    return {
        "card_type": str(raw_data.get("card_type", "")).strip().capitalize(),
        "name": normalize_name(raw_data.get("name")),
        "aadhaar_number": normalize_aadhaar(raw_data.get("aadhaar_number")),
        "pan_number": normalize_pan(raw_data.get("pan_number")),
        "dob": normalize_dob(raw_data.get("dob")),
        "gender": normalize_gender(raw_data.get("gender")),
        "is_valid_id": bool(raw_data.get("is_valid_id", True))
    }

def compute_sha256(data_bytes: bytes) -> str:
    """Generates cryptographic SHA-256 hash for document integrity and fraud check."""
    return hashlib.sha256(data_bytes).hexdigest()

def verify_document_integrity(file_bytes: bytes, registry_db: List[Dict[str, Any]]) -> Dict[str, Any]:
    """
    Checks if the uploaded file's SHA-256 hash already exists in the system registry
    to prevent document forgery and duplicate loan applications.
    """
    file_hash = compute_sha256(file_bytes)
    match = next((item for item in registry_db if item.get("sha256_hash") == file_hash), None)

    if match:
        return {
            "is_original": False,
            "status": "DUPLICATE_DETECTED",
            "message": "✗ Document already exists in database. Duplicate submissions prohibited.",
            "sha256_hash": file_hash,
            "matched_record": match
        }
    return {
        "is_original": True,
        "status": "VERIFIED_ORIGINAL",
        "message": "✓ Document integrity verified. Original digital fingerprint registered.",
        "sha256_hash": file_hash
    }

def calculate_reducing_balance_emi(principal: float, annual_rate: float, tenure_months: int) -> float:
    """
    Standard Reducing-Balance Equated Monthly Installment (EMI) Formula:
    EMI = P * r * (1 + r)^n / ((1 + r)^n - 1)
    where:
      P = Principal loan amount
      r = Monthly interest rate (annual_rate / (12 * 100))
      n = Tenure in months
    """
    if principal <= 0 or tenure_months <= 0:
        return 0.0
    if annual_rate <= 0:
        return round(principal / tenure_months, 2)

    monthly_rate = annual_rate / (12 * 100)
    pow_factor = math.pow(1 + monthly_rate, tenure_months)
    emi = principal * monthly_rate * pow_factor / (pow_factor - 1)
    return round(emi, 2)

def generate_amortization_schedule(principal: float, annual_rate: float, tenure_months: int) -> List[Dict[str, Any]]:
    """
    Generates month-by-month reducing-balance amortization schedule
    computing opening balance, EMI, principal breakdown, interest breakdown,
    closing balance, and cumulative metrics.
    """
    if principal <= 0 or tenure_months <= 0:
        return []

    emi = calculate_reducing_balance_emi(principal, annual_rate, tenure_months)
    monthly_rate = (annual_rate / (12 * 100)) if annual_rate > 0 else 0.0

    schedule = []
    current_balance = round(float(principal), 2)
    cumulative_interest = 0.0
    cumulative_principal = 0.0

    for month in range(1, tenure_months + 1):
        if current_balance <= 0:
            break

        interest_portion = round(current_balance * monthly_rate, 2)

        if month == tenure_months or (emi - interest_portion) >= current_balance:
            principal_portion = current_balance
            month_emi = round(principal_portion + interest_portion, 2)
            closing_balance = 0.0
        else:
            principal_portion = round(emi - interest_portion, 2)
            month_emi = emi
            closing_balance = round(max(current_balance - principal_portion, 0.0), 2)

        cumulative_interest = round(cumulative_interest + interest_portion, 2)
        cumulative_principal = round(cumulative_principal + principal_portion, 2)

        schedule.append({
            "month": month,
            "opening_balance": current_balance,
            "emi": month_emi,
            "principal_portion": principal_portion,
            "interest_portion": interest_portion,
            "closing_balance": closing_balance,
            "cumulative_interest": cumulative_interest,
            "cumulative_principal": cumulative_principal
        })
        current_balance = closing_balance

    return schedule

def process_repayment_waterfall(
    outstanding_balance: float,
    current_repaid: float,
    current_interest_paid: float,
    annual_rate: float,
    payment_amount: float,
    payment_type: str = "emi",
    penalty_due: float = 0.0
) -> Dict[str, Any]:
    """
    Allocates repayments hierarchically using waterfall settlement:
      Priority 1: Late Fees & Penalties
      Priority 2: Accrued Interest on Outstanding Balance
      Priority 3: Principal Reduction / Prepayment Foreclosure
    Updates total repaid, total interest, remaining balance, and loan status.
    """
    if payment_amount <= 0:
        raise ValueError("Payment amount must be greater than zero.")

    rem = payment_amount
    penalty_settled = 0.0
    interest_settled = 0.0
    principal_settled = 0.0

    if penalty_due > 0:
        penalty_settled = min(rem, penalty_due)
        rem -= penalty_settled

    monthly_rate = (annual_rate / (12 * 100)) if annual_rate > 0 else 0.0
    accrued_monthly_interest = round(outstanding_balance * monthly_rate, 2)

    p_type = payment_type.lower()
    if p_type in ("emi", "emi payment"):
        if rem <= accrued_monthly_interest:
            interest_settled = rem
            principal_settled = 0.0
        else:
            interest_settled = accrued_monthly_interest
            principal_settled = min(round(rem - interest_settled, 2), outstanding_balance)

    elif p_type == "interest":
        interest_settled = rem
        principal_settled = 0.0

    elif p_type in ("principal", "prepayment", "foreclosure"):
        principal_settled = min(rem, outstanding_balance)
        interest_settled = 0.0

    new_balance = max(round(outstanding_balance - principal_settled, 2), 0.0)
    new_total_repaid = round(current_repaid + principal_settled, 2)
    new_total_interest = round(current_interest_paid + interest_settled, 2)
    new_status = "Closed" if new_balance <= 0 else "Active"

    return {
        "payment_type": payment_type,
        "amount_received": payment_amount,
        "penalty_settled": penalty_settled,
        "interest_settled": interest_settled,
        "principal_settled": principal_settled,
        "previous_balance": outstanding_balance,
        "new_balance": new_balance,
        "total_principal_repaid": new_total_repaid,
        "total_interest_paid": new_total_interest,
        "status": new_status,
        "settled_at": datetime.now(timezone.utc).isoformat()
    }

def match_names(user_name: str, extracted_name: str) -> bool:
    """
    Performs deterministic tokenized and fuzzy name matching,
    accounting for initials, reverse name order, and token subsets.
    """
    u = normalize_name(user_name)
    e = normalize_name(extracted_name)
    if not u or not e:
        return False
    if u == e:
        return True

    u_tokens = set(u.split())
    e_tokens = set(e.split())

    if u_tokens == e_tokens:
        return True
    if u_tokens.issubset(e_tokens) or e_tokens.issubset(u_tokens):
        return True

    u_long = {w for w in u_tokens if len(w) > 1}
    e_long = {w for w in e_tokens if len(w) > 1}
    if u_long and e_long and (u_long == e_long or u_long.issubset(e_long) or e_long.issubset(u_long)):
        return True

    return (u in e) or (e in u)

def match_dobs(user_dob: str, extracted_dob: str) -> bool:
    """Matches dates of birth with tolerance for year-only representations."""
    if not user_dob or not extracted_dob:
        return True
    u_comps = parse_date_components(user_dob)
    e_comps = parse_date_components(extracted_dob)

    if not u_comps or not e_comps:
        return normalize_dob(user_dob) == normalize_dob(extracted_dob)

    if u_comps[1] is None or e_comps[1] is None:
        return u_comps[0] == e_comps[0]

    return u_comps == e_comps

def run_deterministic_kyc_verification(
    user_input: Dict[str, Any],
    aadhaar_extracted: Dict[str, Any],
    pan_extracted: Optional[Dict[str, Any]] = None
) -> Dict[str, Any]:
    """
    Deterministic matching between user input application fields
    and OCR extracted/sanitized document fields for BOTH Aadhaar and PAN cards.
    Both documents are strictly compulsory for loan verification approval.
    """
    # Support passing a combined dict {"aadhaar": ..., "pan": ...} in second argument
    if pan_extracted is None:
        if isinstance(aadhaar_extracted, dict) and "aadhaar" in aadhaar_extracted and "pan" in aadhaar_extracted:
            pan_extracted = aadhaar_extracted["pan"]
            aadhaar_extracted = aadhaar_extracted["aadhaar"]
        else:
            pan_extracted = {}

    aadhaar_data = aadhaar_extracted or {}
    pan_data = pan_extracted or {}

    u_name = user_input.get("name", "")
    a_name = aadhaar_data.get("name", "")
    p_name = pan_data.get("name", "")

    aadhaar_name_matched = match_names(u_name, a_name)
    pan_name_matched = match_names(u_name, p_name)
    cross_name_matched = match_names(a_name, p_name)
    name_matched = bool(aadhaar_name_matched and pan_name_matched and cross_name_matched)

    u_dob = user_input.get("dob", "")
    a_dob = aadhaar_data.get("dob", "")
    p_dob = pan_data.get("dob", "")

    aadhaar_dob_matched = match_dobs(u_dob, a_dob)
    pan_dob_matched = match_dobs(u_dob, p_dob)
    cross_dob_matched = match_dobs(a_dob, p_dob)
    dob_matched = bool(aadhaar_dob_matched and pan_dob_matched and cross_dob_matched)

    u_gen = normalize_gender(user_input.get("gender", ""))
    a_gen = normalize_gender(aadhaar_data.get("gender", ""))
    gender_matched = (u_gen == a_gen) if (u_gen and a_gen) else True

    u_aadhaar = normalize_aadhaar(user_input.get("aadhaar_number", ""))
    e_aadhaar = normalize_aadhaar(aadhaar_data.get("aadhaar_number", ""))
    aadhaar_number_matched = bool(
        u_aadhaar and e_aadhaar and (u_aadhaar == e_aadhaar or (len(e_aadhaar) == 4 and u_aadhaar[-4:] == e_aadhaar))
    )

    u_pan = normalize_pan(user_input.get("pan_number", ""))
    e_pan = normalize_pan(pan_data.get("pan_number", ""))
    pan_number_matched = bool(u_pan and e_pan and (u_pan == e_pan))

    aadhaar_valid = bool(aadhaar_data.get("is_valid_id", True) and e_aadhaar)
    pan_valid = bool(pan_data.get("is_valid_id", True) and e_pan)
    both_documents_present = bool(e_aadhaar and e_pan)

    checks = {
        "both_documents_present": both_documents_present,
        "aadhaar_present": bool(e_aadhaar),
        "pan_present": bool(e_pan),
        "aadhaar_number_match": aadhaar_number_matched,
        "pan_number_match": pan_number_matched,
        "aadhaar_name_match": aadhaar_name_matched,
        "pan_name_match": pan_name_matched,
        "cross_document_name_match": cross_name_matched,
        "aadhaar_dob_match": aadhaar_dob_matched,
        "pan_dob_match": pan_dob_matched,
        "cross_document_dob_match": cross_dob_matched,
        "gender_match": gender_matched,
        "aadhaar_valid": aadhaar_valid,
        "pan_valid": pan_valid
    }

    is_verified = bool(
        both_documents_present and
        aadhaar_number_matched and
        pan_number_matched and
        name_matched and
        dob_matched and
        gender_matched and
        aadhaar_valid and
        pan_valid
    )

    if not both_documents_present:
        status = "REJECTED_MISSING_DOCUMENT"
    elif is_verified:
        status = "APPROVED"
    else:
        status = "REJECTED_MISMATCH"

    return {
        "is_verified": is_verified,
        "status": status,
        "both_documents_compulsory": True,
        "checks": checks,
        "verified_fields": {
            "name": {
                "user": u_name,
                "extracted_aadhaar": a_name,
                "extracted_pan": p_name,
                "matched": name_matched
            },
            "dob": {
                "user": u_dob,
                "extracted_aadhaar": a_dob,
                "extracted_pan": p_dob,
                "matched": dob_matched
            },
            "gender": {
                "user": u_gen,
                "extracted_aadhaar": a_gen,
                "matched": gender_matched
            },
            "aadhaar_number": {
                "user": u_aadhaar,
                "extracted": e_aadhaar,
                "matched": aadhaar_number_matched
            },
            "pan_number": {
                "user": u_pan,
                "extracted": e_pan,
                "matched": pan_number_matched
            }
        }
    }

def execute_loan_multi_stage_pipeline(
    aadhaar_bytes: Optional[bytes] = None,
    pan_bytes: Optional[bytes] = None,
    user_application: Optional[Dict[str, Any]] = None,
    document_registry: Optional[List[Dict[str, Any]]] = None,
    raw_document_bytes: Optional[bytes] = None,
    card_type: Optional[str] = None
) -> Dict[str, Any]:
    """
    Executes the multi-stage aggregation pipeline chaining all 6 algorithms
    from compulsory dual-document upload (Aadhaar & PAN) to structured verification and amortization.
    """
    if document_registry is None:
        document_registry = []

    # Handle legacy single-document call if passed as (raw_document_bytes, user_application, ...)
    if aadhaar_bytes is not None and isinstance(pan_bytes, dict) and user_application is None:
        user_application = pan_bytes
        pan_bytes = None

    if aadhaar_bytes is None and raw_document_bytes is not None:
        if card_type and card_type.lower() == "pan":
            pan_bytes = raw_document_bytes
        else:
            aadhaar_bytes = raw_document_bytes

    stages_log = []
    stages_log.append("Stage 1: Document Ingestion & Validation - Verifying compulsory Aadhaar & PAN documents.")

    if not aadhaar_bytes or not pan_bytes:
        missing_docs = []
        if not aadhaar_bytes:
            missing_docs.append("Aadhaar Card")
        if not pan_bytes:
            missing_docs.append("PAN Card")
        error_msg = f"✗ Verification rejected: {', '.join(missing_docs)} is missing. Both Aadhaar and PAN documents are strictly compulsory."
        stages_log.append(f"Stage 1 FAILED: {error_msg}")
        return {
            "pipeline_success": False,
            "error_stage": "Stage 1: Document Ingestion",
            "message": error_msg,
            "stages_log": stages_log,
            "final_approval_status": "REJECTED_MISSING_DOCUMENT"
        }

    stages_log.append("Stage 1: Ingested both Aadhaar and PAN documents successfully.")

    aadhaar_integrity = verify_document_integrity(aadhaar_bytes, document_registry)
    pan_integrity = verify_document_integrity(pan_bytes, document_registry)
    stages_log.append(f"Stage 2: SHA-256 Data Integrity -> Aadhaar: {aadhaar_integrity['status']}, PAN: {pan_integrity['status']}")

    if not aadhaar_integrity["is_original"] or not pan_integrity["is_original"]:
        failed_doc = "Aadhaar" if not aadhaar_integrity["is_original"] else "PAN"
        failed_res = aadhaar_integrity if not aadhaar_integrity["is_original"] else pan_integrity
        stages_log.append(f"Stage 2 FAILED: Duplicate or fraudulent {failed_doc} document detected.")
        return {
            "pipeline_success": False,
            "error_stage": "Stage 2: SHA-256 Data Integrity",
            "failed_document": failed_doc,
            "integrity_result": failed_res,
            "document_integrity": {
                "aadhaar": aadhaar_integrity,
                "pan": pan_integrity
            },
            "stages_log": stages_log,
            "final_approval_status": "REJECTED_FRAUD_DUPLICATE"
        }

    raw_aadhaar = run_gemini_vision_ocr(aadhaar_bytes, card_type="Aadhaar")
    raw_pan = run_gemini_vision_ocr(pan_bytes, card_type="PAN")
    stages_log.append(f"Stage 3: Gemini Vision OCR executed -> Aadhaar ({raw_aadhaar.get('ocr_engine')}), PAN ({raw_pan.get('ocr_engine')})")

    sanitized_aadhaar = sanitize_extracted_payload(raw_aadhaar)
    sanitized_pan = sanitize_extracted_payload(raw_pan)
    stages_log.append("Stage 4: Regex-Based String Normalization & Sanitization executed for both documents.")

    kyc_result = run_deterministic_kyc_verification(user_application, sanitized_aadhaar, sanitized_pan)
    stages_log.append(f"Stage 5: Dual Deterministic KYC Matching -> Status: {kyc_result['status']}")

    loan_amount = float(user_application.get("amount_requested", 500000))
    annual_rate = float(user_application.get("interest_rate", 10.5))
    tenure_months = int(user_application.get("tenure_months", 12))

    emi = calculate_reducing_balance_emi(loan_amount, annual_rate, tenure_months)
    amortization_schedule = generate_amortization_schedule(loan_amount, annual_rate, tenure_months)
    stages_log.append(f"Stage 6: EMI Reducing-Balance computed: ₹{emi:,.2f} over {tenure_months} months.")

    repayment_simulation = process_repayment_waterfall(
        outstanding_balance=loan_amount,
        current_repaid=0.0,
        current_interest_paid=0.0,
        annual_rate=annual_rate,
        payment_amount=emi,
        payment_type="emi"
    )
    stages_log.append("Stage 7: Loan Repayment Waterfall & Settlement simulated.")

    structured_data = {
        "pipeline_success": True,
        "stages_log": stages_log,
        "document_integrity": {
            "aadhaar": aadhaar_integrity,
            "pan": pan_integrity,
            "all_verified": aadhaar_integrity["is_original"] and pan_integrity["is_original"]
        },
        "ocr_extracted": {
            "aadhaar": sanitized_aadhaar,
            "pan": sanitized_pan
        },
        "kyc_verification": kyc_result,
        "financial_underwriting": {
            "principal_requested": loan_amount,
            "annual_interest_rate_pct": annual_rate,
            "tenure_months": tenure_months,
            "monthly_emi": emi,
            "total_repayable": round(emi * tenure_months, 2),
            "total_interest_payable": round((emi * tenure_months) - loan_amount, 2),
            "amortization_schedule": amortization_schedule[:3]
        },
        "repayment_settlement_sample": repayment_simulation,
        "final_approval_status": "APPROVED" if kyc_result["is_verified"] else "MANUAL_REVIEW_REQUIRED"
    }

    return structured_data

def run_portfolio_aggregation_pipeline(loan_records: List[Dict[str, Any]]) -> Dict[str, Any]:
    """
    Simulates multi-stage aggregation pipeline ($match, $group, $project)
    to calculate portfolio metrics across all loan facilities.
    """
    df = pd.DataFrame(loan_records)
    if df.empty:
        return {"total_loans": 0, "total_sanctioned": 0.0}

    total_loans = len(df)
    total_sanctioned = float(df["amount_requested"].sum())
    total_disbursed = float(df.get("total_disbursed", pd.Series([0]*total_loans)).sum())
    total_repaid = float(df.get("total_repaid", pd.Series([0]*total_loans)).sum())
    total_outstanding = float(df.get("outstanding_balance", df["amount_requested"]).sum())

    status_breakdown = df["status"].value_counts().to_dict()

    return {
        "portfolio_summary": {
            "total_applications": total_loans,
            "total_sanctioned_amount": round(total_sanctioned, 2),
            "total_disbursed_amount": round(total_disbursed, 2),
            "total_principal_repaid": round(total_repaid, 2),
            "total_active_outstanding": round(total_outstanding, 2),
            "utilization_rate_pct": round((total_disbursed / total_sanctioned * 100), 2) if total_sanctioned > 0 else 0.0
        },
        "status_distribution": status_breakdown
    }

def load_single_document(
    card_type: str,
    default_name: str = "SAMPLE APPLICANT",
    default_number: str = "",
    default_dob: str = "DD/MM/YYYY",
    default_gender: str = "MALE"
) -> Tuple[bytes, str]:
    """
    Loads or uploads a single identity document (Aadhaar or PAN).
    Supports Google Colab upload widget, local file path entry, or synthetic test generation.
    """
    doc_bytes = None
    filename = ""

    if "google.colab" in sys.modules:
        print(f"\n[Compulsory] Please upload your {card_type} card file (JPG, PNG, or PDF):")
        try:
            from google.colab import files
            uploaded = files.upload()
            if uploaded:
                filename = list(uploaded.keys())[0]
                doc_bytes = uploaded[filename]
                print(f"✓ Uploaded {card_type} card '{filename}' successfully ({len(doc_bytes):,} bytes).")
        except Exception as e:
            print(f"Upload error: {e}")

    if not doc_bytes:
        file_path = input(f"Enter local path to {card_type} card file (or press Enter for test sample): ").strip()
        if file_path and os.path.exists(file_path):
            with open(file_path, "rb") as f:
                doc_bytes = f.read()
            filename = os.path.basename(file_path)
            print(f"✓ Loaded {card_type} card '{filename}' ({len(doc_bytes):,} bytes).")
        else:
            print(f"Generating synthetic test {card_type} card...")
            doc_bytes = create_sample_identity_card(
                card_type=card_type,
                name=default_name,
                doc_number=default_number,
                dob=default_dob,
                gender=default_gender
            )
            filename = f"sample_{card_type.lower()}_card.jpg"
            print(f"✓ Generated test {card_type} card '{filename}' ({len(doc_bytes):,} bytes).")

    return doc_bytes, filename
